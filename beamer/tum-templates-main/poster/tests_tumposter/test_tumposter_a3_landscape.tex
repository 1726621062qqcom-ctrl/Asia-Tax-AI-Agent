% .:: Example poster with 3 columns
% ----------------------------------------------------------------------------

\TUMpostersetup{
  columns = 3,
  rows = 4,
  boxes = TUMBoxBlank,
  printaffil = false,
  showframe = true,
}
\maketitle
\begin{tcbposter}
  % footer with some contact information
  \posterbox[fontupper=\footnotesize]{
    name=footer, column=1, span=3, above=bottom
  }{
    \begin{tabularx}{\textwidth}{p{.3\textwidth}XXX}
      \textbf{\theUniversityName}\newline\theSchoolName
       &
      \textbf{First Author}\newline Some contact information\newline
      E-mail: \href{mailto:first.author@tum.de}{first.author@tum.de}
       &
      \textbf{Dr. Second Author}\newline Don't contact me!
       &
      \textbf{Prof. Dr. Last Author}\newline Arcisstr. 21, 80333 Munich (D)
      \newline E-mail: \href{mailto:last.author@tum.de}{last.author@tum.de}
      \newline \href{www.ei.tum.de}{www.ei.tum.de}
    \end{tabularx}
  }

  % figure on the top right
  \posterbox{name=fig1, column=2, below=top}{
    \includegraphics[width=\columnwidth, height=5cm]{example-image-a}
    \captionof{figure}{This is a figure caption. The figure contains a
      placeholder to test the layout.}
  }

  % main text box is a sequence over the rest of the page
  \posterbox[title=The One and Only box]{name=text,
    sequence = 1 between top and footer then
    2 between fig1 and footer then
    3 between top and footer
  }{
    \lipsum[1-4]
  }
\end{tcbposter}


% .:: Example poster with 2 columns
% ----------------------------------------------------------------------------

\clearpage
\TUMpostersetup{
  columns = 2,
  rows = 2,
  boxes = TUMBoxBlank,
  printaffil = true,
  showframe = false,
}
\maketitle
\begin{tcbposter}
  \newcommand{\figheight}{0.8\dimexpr\tcbtextheight\relax}
  \newcommand{\figwidth}{0.8\dimexpr\tcbtextwidth\relax}

  % figure on the top right
  \posterbox{name=fig1, column=2, row=1}{
    \centering
    \includegraphics[width=\figwidth, height=\figheight]{example-image-a}
    \captionof{figure}{This is a figure caption. The figure contains a
      placeholder to test the layout.}
  }

  % references box
  \posterbox[fontupper=\small]{name=refs, column=2, above=bottom}{
    \renewcommand{\section}[2]{}% don't print header
    \nocite{*}% insert all publications (even if they are not cited)
    \bibliographystyle{unsrt}
    \bibliography{sample}
  }

  % main text box is a sequence over the rest of the page
  \posterbox{name=text,
    sequence = 1 between top and bottom then 2 between fig1 and refs
  }{
    \lipsum[1-3]
  }
\end{tcbposter}

\end{document}
