# 伪代码

**定位**：`engine.py`、`ex_ante.py`、`ex_post.py` 的逻辑骨架，2026-10-07 已按此实现在 `tax_graph/`（规则来自 [规则层.md](规则层.md)，待定组的相关性判据见该文 §5）。下文的原型输出是占位税率下的结果，真实数字见 `python -m tax_graph.check_engine`。类型见 [图模型.md](图模型.md)。

## 1. 验证结论

1. **`U` 是返回值，不是停止。** 条件返回 `U` 时计算照常进行，税负返回区间 `(lo, hi)`：`hi` 按不利方向，`lo` 按有利方向。
2. **停止只有两种异常。** `Reject`：入口核心字段不全。`Defect`：断言不过。
3. **已知值短路掉 `U`。** `AND(F, U) == F`。持股比例已知不达标时，受益所有人的字段一个都不用收。所以 `needs` 是 `required_fields()` 的子集。
4. **排序单位是组。** 合取条件里单补一个字段不起作用。一条规则的 `needs` 是一组，排序键是 `cost[g]`，即失去这一组的代价。`cost` 不可相加。
5. **`U` 有三个来源，同一种处理。** `MISSING` 进 `needs`；`PLAN`、`DISCRETION` 进 `conditions`。
6. **`finalize` 是代入重算。** `ticks` 当作已知值代入，重算后取最小。

原型输出：

| 调用 | `(lo, hi)` | 待定组与 `cost` |
|---|---|---|
| `ex_ante`，`holdco_loc=None` | `(50, 100)` | 股息减免 50 |
| `ex_ante`，`holdco_loc=HK` | `(50, 165)` | 境外收入加征 115；股息减免 50 |
| `finalize`，`holdco_loc=None`，全部勾选可满足 | `(50, 50)` | 无 |
| `ex_post`，`actual=100`，证据未齐 | `(50, 100)` | `confirmed=0`，`potential=50` |
| `ex_post`，证据补齐 | `(50, 50)` | `confirmed=50` |

`holdco_loc=HK` 的 `lo` 相同、`hi` 更高、条件更多，被 `None` 支配。

## 2. 值、异常、三值逻辑

```python
T, F, U = "T", "F", "U"

@dataclass(frozen=True)
class Need:
    source: Source                      # MISSING | PLAN | DISCRETION
    key: str

@dataclass(frozen=True)
class Tri:
    v: str
    needs: frozenset[Need] = frozenset()

class Reject(Exception): ...            # 入口：核心字段不全
class Defect(Exception): ...            # 内部：断言不过

def AND(*xs):
    if any(x.v == F for x in xs):
        return Tri(F)                   # 已知 F 使其余 U 无关
    us = [x for x in xs if x.v == U]
    return Tri(U, union(x.needs for x in us)) if us else Tri(T)

def OR(*xs):
    if any(x.v == T for x in xs):
        return Tri(T)
    us = [x for x in xs if x.v == U]
    return Tri(U, union(x.needs for x in us)) if us else Tri(F)

def NOT(x):
    return Tri({T: F, F: T, U: U}[x.v], x.needs)
```

## 3. 谓词

```python
class Ctx:
    mode: Mode                          # EX_ANTE | EX_POST
    facts: FactPack
    ticks: dict[Need, bool]

    def test(self, elem, field, pred) -> Tri:
        if self.mode is EX_ANTE and field in ADJUSTABLE:
            n = Need(PLAN, f"{elem}.{field}:{pred.__name__}")
            return tri(self.ticks[n]) if n in self.ticks else Tri(U, {n})
        if not self.facts.has(elem, field) or not self.facts.evidence(elem, field):
            return Tri(U, {Need(MISSING, f"{elem}.{field}")})
        return tri(pred(self.facts.get(elem, field)))

    def discretion(self, elem, name) -> Tri:
        n = Need(DISCRETION, f"{elem}.{name}")
        return tri(self.ticks[n]) if n in self.ticks else Tri(U, {n})
```

```python
def cond(ctx, elem, c) -> Tri:                  # Leaf → ctx.test(elem, field, pred(P[param]))；Discretion → ctx.discretion
def decide(tri, favourable, bound, group, force) -> bool:
    if tri.v != U:
        return tri.v == T
    if group in force:
        return favourable if force[group] == "fav" else not favourable
    return favourable if bound is FLOOR else not favourable

def engine(graph, ctx, bound, force={}) -> tuple[Money, Groups, Labels]:
    groups, L = {}, {}
    for f in topo_order(graph.flows):           # 按资金流向的依赖序
        items = []
        for stage in STAGES:                    # CHARGE, BASE, RATE_CAP, EXEMPT；阶段内顺序无关
            for r in match(RULES, f, stage, on=f.on):
                tri = cond(ctx, f, r.cond)
                g = (r.id, f.id)
                if tri.v == U:
                    groups[g] = tri.needs
                if decide(tri, r.favourable, bound, g, force) and not overridden(r, ctx, f, bound):
                    items = apply(items, r, P)  # 每个 TaxItem 带 cites = r.sources
        L[f.id] = Label(items, cites(items))
    for e in graph.entities:
        L[e.id] = relieve_double_tax(e, L)      # ENTITY_STAGES：CREDIT, DEFER
    check_citations(L); check_invariants(L)     # 否则 raise Defect
    return total(L), groups, L
```

## 5. 区间与待定组的代价

```python
def bounds(graph, ctx) -> Bounds:
    lo, groups, L_lo = engine(graph, ctx, FLOOR)
    hi, _,      L_hi = engine(graph, ctx, CEILING)
    if lo > hi:
        raise Defect("floor above ceiling")
    cost = {}
    for g in groups:
        cost[g] = engine(graph, ctx, FLOOR, {g: "unfav"})[0] - lo      # 失去这一组的代价
        if engine(graph, ctx, CEILING, {g: "fav"})[0] > hi:
            raise Defect("non-monotone rule interaction")
    return Bounds(lo, hi, groups, cost, L_lo, L_hi)
```

## 6. ex_ante 与 finalize

```python
def rank(row):
    return (row.lo, row.hi, row.n_conditions)

def ex_ante(event, facts) -> ExAnteOutput:
    validate_core(facts)                                    # 否则 raise Reject
    base = build_graph(facts)
    ctx  = Ctx(EX_ANTE, facts, ticks={})
    table = [Row(c, bounds(apply_edits(base, c), ctx))      # 可并行
             for c in candidates(DOMAIN, event)]
    table.sort(key=rank)
    check_metamorphic(ex_ante, event, facts)
    check_roundtrip(table[0])
    return ExAnteOutput(
        table,
        needs      = by_source(table, {MISSING}),           # 按 cost 降序
        conditions = by_source(table, {PLAN, DISCRETION}))

def finalize(table, ticks) -> Row:                          # 人工之后
    ctx = Ctx(EX_ANTE, facts, ticks)
    return min((Row(r.cand, bounds(apply_edits(base, r.cand), ctx)) for r in table), key=rank)
```

## 7. ex_post

```python
def ex_post(facts, actual, t0, ticks={}) -> ExPostOutput:
    validate_core(facts)
    b = bounds(build_graph(facts), Ctx(EX_POST, facts, ticks))
    claims, archived = [], []
    for e in actual:
        if t0 > actual[e].paid_on + rule_of(e).limit:
            archived.append(e); continue
        confirmed = max(0, actual[e].tax - b.L_hi[e].tax)
        potential = max(0, actual[e].tax - b.L_lo[e].tax)
        if confirmed > actual[e].tax:
            raise Defect("refund above tax paid")
        if potential > 0:
            claims.append(Claim(e, confirmed, potential, PATHS[kind_of(e), mount_of(e)]))
    claims.sort(key=lambda c: c.potential, reverse=True)
    return ExPostOutput(claims, needs=sorted(b.groups, key=b.cost.get, reverse=True), archived=archived)
```

人工把做不足的组勾为 `False`，等于把该组的条件置为 `F`，再调一次 `ex_post(..., ticks)`。
